cmake_minimum_required(VERSION 3.20)
project(WgcWinRTCapture CXX CPP)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# C++/WinRT 路径
set(CPPWINRT_PATH "C:\\Program Files (x86)\\Windows Kits\\10\\Include\\10.0.26100.0\\winrt")

# Windows SDK 路径
set(WINDOWS_SDK_PATH "C:\\Program Files (x86)\\Windows Kits\\10")

# 包含目录
include_directories(
    "${CPPWINRT_PATH}"
    "${WINDOWS_SDK_PATH}\\Include\\10.0.26100.0\\shared"
    "${WINDOWS_SDK_PATH}\\Include\\10.0.26100.0\\um"
)

# 库目录
link_directories(
    "${WINDOWS_SDK_PATH}\\Lib\\10.0.26100.0\\um\\x64"
)

# 源文件
set(SOURCES
    wgc_winrt.cpp
)

# 头文件
set(HEADERS
    wgc_winrt.h
)

# 创建 DLL
add_library(wgc_winrt SHARED ${SOURCES} ${HEADERS})

# 链接库
target_link_libraries(wgc_winrt
    d3d11.lib
    dxgi.lib
    dwmapi.lib
    user32.lib
    windowsapp.lib
    WindowsApp.lib  # C++/WinRT
)

# 编译选项
target_compile_options(wgc_winrt PRIVATE
    /wd5104  # 忽略 C++/WinRT 警告
    /await   # 支持 coroutines
)

# 定义导出宏
target_compile_definitions(wgc_winrt PRIVATE
    WGC_WINRT_EXPORTS
    WINRT_LEAN_AND_MEAN
)

# 链接时优化
if(MSVC)
    set_target_properties(wgc_winrt PROPERTIES
        LINK_FLAGS "/LTCG"  # 链接时代码生成
    )
endif()
